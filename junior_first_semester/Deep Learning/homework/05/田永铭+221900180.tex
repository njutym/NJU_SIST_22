	\begin{prob}
		\textbf{列举几种常见的激活函数，并简要说明它们的作用。}
	\end{prob}
	
	\begin{sol}
	解：
\begin{enumerate}
	\item \textbf{Sigmoid 函数} \\
	\textit{公式：} \( \sigma(x) = \frac{1}{1 + e^{-x}} \) \\
	\textit{作用：} 将输入值映射到 0 和 1 之间。适用于二分类问题的输出层，因为其输出可解释为概率。但在深层网络中容易导致梯度消失问题，因此不常用于隐藏层。
	
	\item \textbf{Tanh（双曲正切）函数} \\
	\textit{公式：} \( \tanh(x) = \frac{e^{x} - e^{-x}}{e^{x} + e^{-x}} \) \\
	\textit{作用：} 输出值在 -1 和 1 之间，比 Sigmoid 有更好的性能，因为其输出是零中心的，有助于加速梯度下降。与 Sigmoid 类似，Tanh 也存在梯度消失的问题。
	
	\item \textbf{ReLU（Rectified Linear Unit）} \\
	\textit{公式：} \( \text{ReLU}(x) = \max(0, x) \) \\
	\textit{作用：} 当输入大于 0 时，输出为输入值；否则输出为 0。ReLU 是目前最常用的激活函数之一，能够有效缓解梯度消失问题，且计算简单。其缺点是会出现“死神经元”问题，导致部分神经元永远不会激活。
	
	\item \textbf{Leaky ReLU} \\
	\textit{公式：} \( \text{Leaky ReLU}(x) = \max(\alpha x, x) \)（其中 \(\alpha\) 是一个小常数） \\
	\textit{作用：} Leaky ReLU 是对 ReLU 的改进，允许输入小于 0 时输出为一个很小的负数，避免了死神经元问题。通常，\(\alpha\) 取值较小，如 0.01。
	
	\item \textbf{GELU (Gaussian Error Linear Unit)} \\
	\textit{公式：} \( \text{GELU}(x) = 0.5 x \left( 1 + \tanh\left( \sqrt{\frac{2}{\pi}} (x + 0.044715 x^3) \right) \right) \) \\
	\textit{作用：} GELU 是一种平滑的非线性激活函数，常用于 Transformer 等模型中。它结合了高斯误差函数和线性激活的优点，使得网络在训练时能够获得更好的梯度流，避免 ReLU 的死神经元问题，并具有类似于 ReLU 的高效性。
\end{enumerate}
	\end{sol}

	\begin{prob}
		\textbf{什么是反向传播？它的核心思想是什么？}
	\end{prob}
	
	\begin{sol}
		解:\\
		\textbf{反向传播（Backpropagation） :}\\是一种用于训练神经网络的算法，它通过链式法则有效地计算损失函数关于网络中每个参数的梯度，从而实现模型的优化。反向传播通过梯度下降算法来更新神经网络中的权重和偏置，使得损失函数的值逐渐减小，从而提高网络的性能。\\
		\textbf{核心思想：}\\
		反向传播的核心思想是利用\textbf{链式法则}，从输出层开始，\textbf{逐层将误差反向传播到隐藏层和输入层}，计算每个参数对损失函数的\textbf{偏导数}，并利用这些梯度更新参数。
	\end{sol}

		解：
		
		
		\textbf{全连接网络（FC）或多层感知机（MLP）}
		
		全连接网络（FC）或多层感知机（MLP）是一种神经网络模型，由多层神经元组成，其中每一层的每个神经元与前一层的每个神经元都有连接。它是最基础的神经网络类型之一，广泛用于分类和回归任务。
		\begin{itemize}
			\item \textbf{输入层}：接收原始输入数据。
			\item \textbf{隐藏层}：通过加权和偏置进行计算，并通过激活函数进行非线性变换。
			\item \textbf{输出层}：根据任务需求输出结果，如分类标签或回归值。
		\end{itemize}
		
		MLP 是一种前馈神经网络，信息在网络中单向传播，不会有反馈连接。每一层的输出都成为下一层的输入。
		
		\textbf{线性分类器}
		
		线性分类器（如线性回归或支持向量机）通过输入特征的线性组合来做出预测，输出一个线性函数，通常表示为：
		\[
		y = w_1 x_1 + w_2 x_2 + \cdots + w_n x_n + b
		\]
		其中 \( x_1, x_2, \dots, x_n \) 是输入特征，\( w_1, w_2, \dots, w_n \) 是对应的权重，\( b \) 是偏置项。
		
		而多层感知机通过多个层级和非线性激活函数的组合，可以拟合更复杂的非线性关系。每一层的输出通过激活函数进行非线性变换，从而具有更强的表达能力。
		
		\textbf{主要区别}
		\begin{itemize}
			\item \textbf{线性与非线性}：
			\begin{itemize}
				\item 线性分类器只能表示输入特征的线性关系。
				\item MLP 使用非线性激活函数，使其能够学习复杂的非线性关系。
			\end{itemize}
			\item \textbf{层次结构}：
			\begin{itemize}
				\item 线性分类器只有一个层次，直接从输入到输出。
				\item MLP 由多层组成，包括至少一个隐藏层。
			\end{itemize}
			\item \textbf{表达能力}：
			\begin{itemize}
				\item 线性分类器的表达能力较弱，一般仅适用于线性可分的任务。
				\item MLP 可以逼近任何复杂的函数，适用于更广泛的任务。
			\end{itemize}
		\end{itemize}
	\end{sol}
	
	\begin{prob}
		\textbf{在梯度流的基本计算模式中，加法门(add gate)、乘法门(mul gate)和最大值门(max gate)分别具有什么特点？}
	\end{prob}
	
	\begin{sol}
		解：\\
		\textbf{加法门：}加法门不改变梯度，只是将梯度传递到所有输入。
		
		\textbf{乘法门：}乘法门会将梯度乘以其他输入的乘积（假设是多个，如果是两个输入的乘法则是另一个输入）传到这个输入。
		
		\textbf{最大值门：}最大值门会将梯度传递给选择出的最大值，但如果最大值有多个，则平分梯度。
